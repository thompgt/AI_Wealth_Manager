# Pinned to exact versions.
#
# These were `>=` constraints, which means two installs a week apart could
# resolve to different dependency trees -- and this stack is unusually
# sensitive to that: yfinance renames `.info` fields and changed its
# auto_adjust default between minor releases, langchain's structured-output
# behaviour moved between majors, and pandas 3 changed semantics the
# analytics code relies on. A "works on my machine" bug in a system that
# sizes positions is not an acceptable class of failure.
#
# Upgrade deliberately: bump a pin, run `pytest`, commit.

# Web / API
fastapi==0.139.0
uvicorn==0.51.0
httpx==0.28.1

# Orchestration
langgraph==1.2.9
langgraph-checkpoint-sqlite==3.1.0   # durable human-in-the-loop checkpoints
langchain-core==1.4.9
langchain-google-genai==4.2.7

# Data / DB
sqlalchemy==2.0.51
psycopg2-binary==2.9.12
pydantic==2.13.4
pydantic-settings==2.14.2
# Schema migrations. create_all can add a table but never alter one, so
# without this any column change is a manual DDL script applied by hand to
# each environment -- and eventually to only some of them.
alembic==1.18.5
email-validator==2.3.0

# Auth
# bcrypt is used directly rather than through passlib: passlib 1.7.4 reads
# bcrypt.__about__, which bcrypt 4.1 removed, so the combination emits errors
# on every hash.
bcrypt==5.0.0
pyjwt==2.12.1

# Observability
prometheus-client==0.24.1

# Analytics
# No PyPortfolioOpt. It was pinned here and never imported -- the Sharpe,
# volatility and drawdown figures are computed directly in agents/diagnostics.py
# and services/performance.py -- while dragging in cvxpy and osqp, a build
# chain that breaks on new Python releases, for nothing.
numpy==2.5.1
pandas==3.0.3
yfinance==1.5.1

# News / research
ddgs==9.14.4
requests==2.34.2
python-dotenv==1.2.2

# Frontend
solara==1.60.3
plotly==6.9.0
# Required by solara.FigurePlotly: plotly 6 moved FigureWidget onto anywidget,
# and neither plotly nor solara declares it. Without it the dashboard renders a
# traceback instead of the results view -- and only on the results view, so a
# smoke test that stops at the client list never sees it.
anywidget==0.11.0
